## PHẦN 1 — GAP-R1: Segment Typing Algorithm Chi tiết

Ba báo cáo đều nêu `NARRATION / DIALOGUE / THOUGHT / SYSTEM / AUTHOR_NOTE` nhưng không có algorithm classify.

### 1.1 Spec đầy đủ

```python
class SegmentType(Enum):
    NARRATION       = "narration"     # Văn miêu tả thuần túy
    DIALOGUE        = "dialogue"      # Lời thoại trực tiếp
    THOUGHT         = "thought"       # Nội tâm nhân vật
    SYSTEM_PROMPT   = "system_ui"     # UI hệ thống 【...】
    CHAPTER_TITLE   = "chapter_title" # Tiêu đề chương
    AUTHOR_NOTE     = "author_note"   # Ghi chú tác giả
    FORUM_POST      = "forum"         # Lời đề từ forum
    ADVERTISEMENT   = "ad"            # Quảng cáo/CTA

class SegmentClassifier:
    """
    Phân loại từng đoạn văn trước khi đưa vào pipeline dịch.
    
    Priority (từ cao đến thấp):
    1. Hard pattern match → immediate classify
    2. Positional heuristics (đầu chương vs thân chương vs cuối chương)
    3. Content heuristics
    4. Fallback: NARRATION
    """

    # ── HARD PATTERNS (P0 — xử lý trước tất cả) ──────────────────
    
    CHAPTER_TITLE_PATTERNS = [
        r"^第[零一二三四五六七八九十百千\d]+[章节回卷篇]",
        r"^Chapter\s+\d+",
        r"^\d+\.\s+\S",
        r"^番外[一二三四五六七八九十]",
    ]

    SYSTEM_UI_PATTERNS = [
        r"^【[^】]{1,100}】$",           # Full bracket line
        r"^〖[^〗]{1,100}〗$",
        r"^「[^」]{1,100}」$",           # Japanese-style bracket
        r"^系统提示[:：]",
        r"^宿主[:：]",
        r"^\[System\]",
    ]

    AUTHOR_NOTE_PATTERNS = [
        r"(?:^|\n)\s*[（(][Pp][Ss][:：。]",
        r"(?:求月票|求推荐票|求收藏|求订阅|求打赏)",
        r"(?:新书上传|新人新书)",
        r"(?:本章完|未完待续|下章预告)",
        r"(?:请假条|作者有话说|作者留言)",
        r"(?:^|\n)\s*\*\*作者[:：]",
        r"感谢.{1,20}打赏",
        r"(?:上架|全订|万订)",
        r"感谢书友.{1,20}(?:月票|推荐票|打赏)",
    ]

    FORUM_POST_PATTERNS = [
        r"(?:楼主|一楼|二楼|沙发|板凳)[:：]",
        r"(?:顶|顶一下|ddd|233+|666+|哈哈哈哈{3,})",
        r"(?:\+1|\-1|赞|同)",
        r"^回复.{1,15}[:：]",
    ]

    DIALOGUE_MARKERS = {
        "open":  ["「", "『", """, "\"", "【", "—"],
        "verbs": ["说", "道", "喊", "叫", "问", "答",
                  "怒道", "笑道", "冷道", "喝道", "低语",
                  "嘀咕", "暗道", "心道", "心想", "沉声道",
                  "轻声道", "惊呼", "嘶吼"],
    }

    THOUGHT_MARKERS = [
        "心中暗道", "心想", "心道", "脑海中", "潜意识",
        "暗自思量", "默默想道", "在心中", "内心深处"
    ]

    # ── POSITIONAL HEURISTICS ─────────────────────────────────────

    CHAPTER_END_NOISE_WINDOW = 50   # chars — last 50 chars of chapter

    def classify(self, segment: str, position: SegmentPosition,
                 chapter_end_distance: int) -> SegmentType:
        """
        position: SegmentPosition(chapter_start=bool, chapter_end=bool)
        chapter_end_distance: chars from end of chapter
        """

        stripped = segment.strip()

        # P0: Hard patterns
        if self._match_any(stripped, self.CHAPTER_TITLE_PATTERNS):
            return SegmentType.CHAPTER_TITLE

        if self._match_any(stripped, self.SYSTEM_UI_PATTERNS):
            return SegmentType.SYSTEM_PROMPT

        if self._match_any(stripped, self.AUTHOR_NOTE_PATTERNS):
            return SegmentType.AUTHOR_NOTE

        if self._match_any(stripped, self.FORUM_POST_PATTERNS):
            return SegmentType.FORUM_POST

        # P1: Positional + pattern
        if chapter_end_distance <= self.CHAPTER_END_NOISE_WINDOW:
            if self._match_any(stripped, self.AUTHOR_NOTE_PATTERNS):
                return SegmentType.AUTHOR_NOTE
            # Loose: short segments at chapter end with PS/求 → author note
            if len(stripped) < 80 and re.search(r"[求谢感]", stripped):
                return SegmentType.AUTHOR_NOTE

        # P2: Content heuristics
        if self._is_dialogue(stripped):
            return SegmentType.DIALOGUE

        if self._is_thought(stripped):
            return SegmentType.THOUGHT

        # Fallback
        return SegmentType.NARRATION

    def _is_dialogue(self, text: str) -> bool:
        """
        Detect dialogue từ:
        1. Opening marker ("「 / "" / —)
        2. Preceding speech verb pattern: X说/X道 + ：/[quote]
        3. Closing marker với speech verb cuối
        """
        has_open_marker = any(text.startswith(m) for m in self.DIALOGUE_MARKERS["open"])
        has_speech_verb = any(v in text[:50] for v in self.DIALOGUE_MARKERS["verbs"])
        return has_open_marker and has_speech_verb

    def _is_thought(self, text: str) -> bool:
        return any(m in text for m in self.THOUGHT_MARKERS)
```

## PHẦN 2 — GAP-R2: ClauseSegmenter Specification

Cả 3 báo cáo dùng `clause_segment()` như black box. Đây là spec chi tiết.

### 2.1 Định nghĩa clause boundary

```python
class ClauseBoundary(Enum):
    SENTENCE_END     = "sentence_end"       # 。！？
    CLAUSE_SOFT      = "clause_soft"        # ，trong điều kiện ngữ pháp
    CLAUSE_HARD      = "clause_hard"        # ；
    QUOTE_OPEN       = "quote_open"         # ：" / 说道
    QUOTE_CLOSE      = "quote_close"        # " / 」
    CONJUNCTION      = "conjunction"        # 但是/而且/所以
    CONDITIONAL      = "conditional"        # 如果/只要/一旦
    CONCESSIVE       = "concessive"         # 虽然/尽管/哪怕

class ClauseSegmenter:
    """
    Tách mệnh đề có awareness về protected spans.
    Nguyên tắc: KHÔNG tách bên trong entity, số, quote, bracket.
    """

    # Discourse frame openers — signal clause boundary BEFORE them
    FRAME_OPENERS = {
        # Conditional
        "如果", "若", "要是", "倘若", "假若", "假设", "假使",
        "只要", "只有", "一旦", "万一",
        "除非", "若非",
        # Concessive
        "虽然", "尽管", "即便", "即使", "哪怕", "就算", "就是",
        # Causal
        "因为", "由于", "既然",
        # Result
        "所以", "因此", "于是", "故而", "从而",
        # Contrast
        "但是", "但", "然而", "却", "不过", "可是", "而",
        # Additive
        "而且", "并且", "同时", "另外", "况且",
        # Sequential
        "然后", "接着", "随后", "之后", "此后",
        # Topic
        "至于", "关于", "对于", "至于说", "说到",
        # Viewpoint
        "作为", "就…而言",
    }

    # Conjunctions that signal boundary AFTER comma
    COMMA_BOUNDARY_TRIGGERS = {
        "但", "却", "而", "然后", "所以", "因此", "于是",
        "不过", "可是", "就算", "哪怕", "即便",
    }

    def segment(self, text: str,
                protected_spans: list[ProtectedSpan]) -> list[Clause]:
        """
        Algorithm:
        1. Mark protected span positions
        2. Scan left to right
        3. At each potential boundary:
           a. Check if in protected span → skip
           b. Check boundary type
           c. Check if next token is frame opener
           d. Decide split or keep
        """
        positions = self._mark_protected(text, protected_spans)
        boundaries = []

        for i, char in enumerate(text):
            if positions[i]:  # In protected span → never split
                continue

            if char in "。！？":
                boundaries.append((i, ClauseBoundary.SENTENCE_END))

            elif char == "；":
                boundaries.append((i, ClauseBoundary.CLAUSE_HARD))

            elif char == "，":
                # Only split comma if next content is frame opener
                lookahead = self._peek_next_token(text, i+1, positions)
                if lookahead in self.FRAME_OPENERS:
                    boundaries.append((i, ClauseBoundary.CONDITIONAL
                                        if lookahead in self.FRAME_OPENERS
                                        else ClauseBoundary.CLAUSE_SOFT))
                elif lookahead in self.COMMA_BOUNDARY_TRIGGERS:
                    boundaries.append((i, ClauseBoundary.CLAUSE_SOFT))
                # else: do not split this comma

        return self._build_clauses(text, boundaries)
```

## PHẦN 3 — GAP-R3: Transliteration Engine — Phoneme Rebuild Spec

V3 nêu `translit engine v1` nhưng không spec rõ.

### 3.1 Algorithm 3 stage đầy đủ

```python
class TransliterationDecisionTree:
    """
    Stage 1: Exact glossary lookup (METHOD C)
    Stage 2: Phoneme-based Latin rebuild (METHOD B)
    Stage 3: Han-Viet fallback (METHOD A)
    """

    def get_vi_form(self, zh_form: str, entity_type: str,
                    register: str, glossary: dict) -> TranslitResult:

        # ── STAGE 1: Exact lookup ─────────────────────────────────
        if zh_form in glossary:
            return TranslitResult(
                vi_form=glossary[zh_form],
                method="glossary",
                confidence=1.0
            )

        # ── Franchise shortcut ─────────────────────────────────────
        for franchise_db in FRANCHISE_DATABASES:
            if zh_form in franchise_db:
                return TranslitResult(
                    vi_form=franchise_db[zh_form],
                    method="franchise_glossary",
                    confidence=0.97
                )

        # ── STAGE 2: Phoneme rebuild ───────────────────────────────
        if self._is_phoneme_transliteration(zh_form):
            raw = self._phoneme_rebuild(zh_form)
            polished = self._post_process_phoneme(raw, entity_type)
            return TranslitResult(
                vi_form=polished,
                method="phoneme_rebuild",
                confidence=0.72
            )

        # ── STAGE 3: Han-Viet ─────────────────────────────────────
        hv_form = self._han_viet_convert(zh_form, register)
        return TranslitResult(
            vi_form=hv_form,
            method="han_viet",
            confidence=0.82
        )

    def _post_process_phoneme(self, raw: str,
                               entity_type: str) -> str:
        """
        Post-processing sau phoneme rebuild:
        1. Capitalize first letter của mỗi word segment
        2. Xử lý double vowels: aei → ai, oua → oa
        3. Xử lý đuôi đặc trưng: -si → -si (giữ), -er → không thêm r
        4. Người tên: Title Case
        5. Địa điểm: Title Case
        6. Tổ chức/faction: Title Case với "Hội/Quân/Liên minh" suffix nếu cần
        """
        # Remove double vowels
        result = re.sub(r"([aeiou])\1+", r"\1", raw)

        # Capitalize
        result = result.title()

        # Entity-type-specific suffix (nếu cần)
        suffix_map = {
            "ORG":    None,       # Giữ nguyên, không thêm suffix
            "PERSON": None,
            "LOC":    None,
            "FACTION": None,
        }
        return result

    def _is_phoneme_transliteration(self, form: str) -> bool:
        """
        Heuristic: Form có phải tên phiên âm nước ngoài sang Hán không?
        
        Indicators:
        1. Tỷ lệ char thuộc HIGH_FREQ_TRANSLIT_SET >= 0.6
        2. Không có nghĩa Hán phổ thông khi ghép lại
        3. Độ dài 2-6 char
        4. Không match common Chinese word patterns
        """
        if not (2 <= len(form) <= 6):
            return False
        if form in COMMON_ZH_WORDS:
            return False
        translit_ratio = sum(
            1 for c in form if c in HIGH_FREQ_TRANSLIT_CHARS
        ) / len(form)
        return translit_ratio >= 0.60
```

### 4.1 Tình trạng hiện tại và rủi ro

```
Hiện tại trong RBMTTranslator:
  output → TM.insert(score=0.92, provenance="rbmt", status="verified")

Rủi ro:
  Lần dịch chương 1: entity sai → vào TM với "verified"
  Lần dịch chương 2: TM lookup tìm thấy → dùng lại bản sai
  Sau 50 chương: lỗi entity khuếch đại, không thể audit
```

```python
class ConflictResolver:
    """
    Khi hai rules cùng claim overlapping span:
    → Rule với priority thấp hơn (số nhỏ hơn) thắng
    → Rule thua bị skip và ghi vào trace
    
    Exception: Entity guard (P1-5) luôn thắng, không exception
    """

    def resolve(self, claims: list[SpanClaim]) -> list[SpanClaim]:
        # Sort by priority ascending
        claims.sort(key=lambda c: c.priority)
        
        accepted = []
        claimed_ranges = []

        for claim in claims:
            overlap = any(
                self._overlaps(claim, existing)
                for existing in claimed_ranges
            )
            if not overlap:
                accepted.append(claim)
                claimed_ranges.append(claim)
            else:
                # Log conflict
                logger.debug(f"Rule {claim.rule_name} (P{claim.priority}) "
                            f"lost span conflict to higher-priority rule")

        return accepted
```

### 6.1 Zero-Pronoun Detection

```python
class ZeroPronounDetector:
    """
    Trong tiếng Trung, zero pronoun (ZP) xuất hiện khi subject/object
    được bỏ qua hoàn toàn. Ví dụ:
    
    ZH: 李宇走进房间。[ZP]看了看四周。
    VI: Lý Vũ bước vào phòng. [Hắn] nhìn xung quanh.
    
    Tiếng Việt thường cần điền đại từ vào, nhất là khi chuyển sang
    câu mới. Nhưng có trường hợp không cần điền (văn học tự nhiên
    hơn khi bỏ chủ ngữ nếu subject rõ ràng).
    """

    # Verbs thường chỉ có một subject duy nhất khả dĩ trong scene
    SINGLE_AGENT_VERBS = {
        "站", "坐", "走", "跑", "看", "想", "说", "叫", "拿",
        "转身", "回头", "低头", "抬头", "皱眉", "点头", "摇头",
    }

    # Verbs thường cần điền ZP rõ ràng trong tiếng Việt
    REQUIRES_EXPLICIT_SUBJECT_VI = {
        "说", "道", "喊", "叫",  # Speech verbs
        "笑", "哭", "叹息",       # Emotion expressions
        "转身", "回头",            # Directional body movement
    }

    def detect_zp_slots(self, zh_sentence: str,
                         entity_memory: EntitySalienceMemory,
                         window: list[str]) -> list[ZeroPronounSlot]:
        """
        Returns danh sách vị trí ZP cần resolve trong câu.
        
        Heuristic:
        1. Câu thiếu explicit subject
        2. Verb trong câu thuộc known verb inventory
        3. Có entity salient gần đây (window 3-5 câu)
        4. Segment type = NARRATION hoặc THOUGHT (không phải DIALOGUE)
        """
        slots = []

        # Simple heuristic: câu bắt đầu bằng VV/ADV (không có NP subject)
        if self._starts_with_verb_or_adv(zh_sentence):
            salient = entity_memory.get_most_salient(top_k=3)
            if salient:
                slots.append(ZeroPronounSlot(
                    position=0,
                    candidates=salient,
                    role="SUBJECT"
                ))

        return slots

    def should_insert_vi(self, slot: ZeroPronounSlot,
                          resolved_entity: EntityRecord,
                          vi_sentence: str) -> bool:
        """
        Quyết định có nên chèn đại từ VI hay không.
        
        CHÈN khi:
        - Entity thay đổi so với câu trước
        - Câu dài/phức tạp và không rõ ràng
        - Verb thuộc REQUIRES_EXPLICIT_SUBJECT_VI
        
        KHÔNG CHÈN khi:
        - Entity giống câu liền trước (tránh lặp đại từ)
        - Văn học Việt tự nhiên bỏ chủ ngữ câu tiếp
        - Segment type là DIALOGUE (thoại tự nhiên)
        
        Nguyên tắc tránh over-insertion:
        Nếu 2 câu liên tiếp có cùng entity → chèn ở câu 1, bỏ câu 2.
        """
        prev_entity = self._get_prev_sentence_entity()
        if prev_entity == resolved_entity:
            return False  # Same entity, no need to repeat
        return True
```

### 7.1 Whitelist Priority System

```python
class NoiseFilterWithSafeWhitelist:
    """
    Nguyên tắc: Whitelist wins over noise score.
    Bất kỳ span nào match whitelist → KEEP, bất kể noise score.
    """

    # Tier 1 Whitelist: HARD KEEP — không bao giờ drop
    TIER1_HARD_KEEP = [
        lambda seg, registry: registry.has_entity(seg),      # Has known entity
        lambda seg, registry: registry.has_system_ui(seg),   # Known system prompt
        lambda seg, _: bool(re.match(r"^第.+章", seg)),      # Chapter title
        lambda seg, registry: registry.is_in_glossary(seg),  # Glossary term
    ]

    # Tier 2 Whitelist: SOFT KEEP — giảm noise score mạnh
    TIER2_SOFT_KEEP = [
        lambda seg, _: len(seg.strip()) > 200,               # Long segment = likely story
        lambda seg, _: any(c in seg for c in "，。！？"),    # Has Chinese punctuation = story
        lambda seg, registry: registry.has_character_name(seg),  # Contains character name
    ]

    # Noise patterns với weight
    NOISE_PATTERNS_WEIGHTED = [
        (r"[（(][Pp][Ss][:：。]", 0.60, "author_ps"),
        (r"求月票|求推荐票|求收藏",  0.70, "vote_request"),
        (r"新书上传|新人新书",       0.80, "promo"),
        (r"本章完$",                 0.50, "chapter_end"),
        (r"未完待续",               0.45, "to_be_continued"),
        (r"作者有话说[:：]",        0.65, "author_note"),
        (r"感谢.{1,30}打赏",        0.55, "donation_thanks"),
        (r"楼主[:：]",              0.60, "forum_post"),
        (r"\[System\]",             0.00, "protected_system"),  # Weight 0 = not noise
    ]

    def filter(self, segment: str,
               registry: ProtectedSpanRegistry,
               seg_type: SegmentType) -> NoiseDecision:

        # TIER 1: Hard whitelist — immediate KEEP
        for check in self.TIER1_HARD_KEEP:
            if check(segment, registry):
                return NoiseDecision.KEEP

        # Already classified AUTHOR_NOTE/ADVERTISEMENT → DROP
        if seg_type in {SegmentType.AUTHOR_NOTE, SegmentType.ADVERTISEMENT}:
            return NoiseDecision.DROP

        # Compute noise score
        score = 0.0
        for pattern, weight, label in self.NOISE_PATTERNS_WEIGHTED:
            if re.search(pattern, segment):
                score += weight

        # TIER 2: Soft whitelist — reduce score
        for check in self.TIER2_SOFT_KEEP:
            if check(segment, registry):
                score *= 0.3  # Reduce noise score by 70%

        # Decision
        if score >= 0.80:
            return NoiseDecision.DROP
        elif score >= 0.45:
            return NoiseDecision.REVIEW
        return NoiseDecision.KEEP
```

```python
class RegressionGate:
    """
    Chặn promotion nếu regression suite fails.
    Mọi rule update/data update phải qua gate này.
    """

    THRESHOLDS = {
        "grammar_transfer": {"min_pass_rate": 0.90},
        "entity_ner":       {"min_f1": 0.88, "min_precision": 0.85},
        "entity_alias":     {"min_accuracy": 0.88},
        "transliteration":  {"min_top1_accuracy": 0.95},
        "noise_filter":     {"min_precision": 0.95, "max_false_drop_rate": 0.02},
        "context_coref":    {"min_accuracy": 0.82},
    }

    def check(self, suite_name: str, results: dict) -> GateResult:
        thresholds = self.THRESHOLDS[suite_name]
        failures = []

        for metric, threshold in thresholds.items():
            actual = results.get(metric)
            if actual is None:
                failures.append(f"Missing metric: {metric}")
            elif actual < threshold:
                failures.append(
                    f"{metric}: {actual:.3f} < required {threshold:.3f}"
                )

        if failures:
            return GateResult(passed=False, failures=failures)
        return GateResult(passed=True)
```

```python
REGISTER_LEXICAL_MAP = {

    # Connectives
    ("但是", "modern"):   "nhưng",
    ("但是", "classical"): "nhưng/song",
    ("然而", "modern"):   "tuy nhiên",
    ("然而", "classical"): "thế nhưng/song",
    ("虽然", "modern"):   "mặc dù",
    ("虽然", "xianxia"):  "tuy",

    # Common verbs — register-specific nuance
    ("死", "modern"):     "chết",
    ("死", "xianxia"):    "chết/viên tịch/ngã xuống",
    ("死", "classical"):  "băng hà/tử trận",
    ("说", "modern"):     "nói",
    ("说", "xianxia"):    "nói/đạo",
    ("说", "classical"):  "nói/viết",
}

class RegisterAwareLexicalDecoder:
    def decode(self, token: str, register: str,
               trie: TrieEngine) -> str:
        # 1. Register-specific override map
        key = (token, register)
        if key in REGISTER_LEXICAL_MAP:
            return REGISTER_LEXICAL_MAP[key]

        # 2. Trie lookup với register filter
        result = trie.lookup(token, register_filter=register)
        if result:
            return result

        # 3. Trie lookup không filter (fallback)
        result = trie.lookup(token)
        if result:
            return result

        # 4. Unresolved
        return f"[[UNRESOLVED:{token}]]"
```

```
  └── 30% → test (holdout, không dùng trong development)
```

```python
    confidence:   float          # Annotator confidence 0-1
    reviewed:     bool
    source_chapter: int
```
