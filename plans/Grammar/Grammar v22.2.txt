```python
class PatternBasedNER:

    # ── TYPE 2: PREFIX-BASED ──────────────────────────────────────
    PREFIX_PATTERNS = {
        # Tên nhân vật có prefix chức danh
        "PERSON_TITLED": [
            (r"(?:舰长|将军|指挥官|议长|族长|团长|队长)([^，。！？\s]{2,6})", 1),
            (r"([^，。！？\s]{2,6})(?:舰长|将军|指挥官)", 1),
            # Xianxia titles
            (r"(?:道友|前辈|大人|阁下|殿下|陛下)\s*([^，。！？\s]{2,6})", 1),
            (r"([^，。！？\s]{2,6})\s*(?:道友|前辈|大人|阁下|殿下|陛下)", 1),
        ],
        # Franchise entity với prefix rõ ràng
        "FRANCHISE_ENTITY": [
            (r"(?:漫威|DC|星战|变形金刚|生化危机)(?:的|中的)?([^，。！？\s]{2,8})", 1),
        ],
        # Tên hành tinh/địa điểm với prefix vị trí
        "PLACE_PREFIXED": [
            (r"(?:前往|到达|位于|来自|出发自)\s*([\u4e00-\u9fff]{2,8}星(?:球|系)?)", 1),
            (r"([\u4e00-\u9fff]{2,8})\s*(?:星系|星域|星区)", 1),
        ],
        # System/tech entity với prefix
        "SYSTEM_ENTITY": [
            (r"(?:使用|激活|装备|携带|持有|获得)\s*【([^】]{2,20})】", 1),
            (r"(?:解锁|升级|点化|进阶)\s*([\u4e00-\u9fff]{2,10})", 1),
        ],
    }

    def _scan_prefix(self, text: str, chapter: int) -> list[EntityCandidate]:
        candidates = []
        for entity_type, patterns in self.PREFIX_PATTERNS.items():
            for pattern, group_idx in patterns:
                for m in re.finditer(pattern, text):
                    name = m.group(group_idx)
                    if len(name) >= 2:
                        candidates.append(EntityCandidate(
                            form=name.strip(),
                            type=entity_type,
                            source="prefix_pattern",
                            chapter=chapter,
                            confidence=0.72
                        ))
        return candidates

    # ── TYPE 6: REPEATED SUBJECT DETECTION ───────────────────────
    # Cụm xuất hiện ≥ 3 lần là subject trong đoạn văn → likely entity
    def _scan_repeated_subject(self, text: str, chapter: int,
                                min_count: int = 3) -> list[EntityCandidate]:
        """
        Tách câu, lấy subject candidate (NP đứng đầu câu trước verb),
        đếm tần suất, nếu ≥ min_count → add candidate.
        """
        sentence_starts = re.findall(
            r"(?:^|[。！？\n])([^，。！？\n]{2,8})(?:是|有|将|会|走|站|看|说|道|喊|叫)",
            text
        )
        counter = Counter(sentence_starts)
        candidates = []
        for form, count in counter.items():
            if count >= min_count and len(form) >= 2:
                candidates.append(EntityCandidate(
                    form=form,
                    type="REPEATED_SUBJECT",
                    source="frequency_subject",
                    chapter=chapter,
                    confidence=0.60 + min(count * 0.05, 0.25)
                ))
        return candidates
```

### B.2 — PASS 3 bổ sung: Negative Context Penalties

```python
class ContextConfirmationPass:

    # Common words không bao giờ là entity
    COMMON_WORD_BLACKLIST = {
        # Verbs
        "知道", "明白", "感觉", "觉得", "认为", "希望", "相信",
        "需要", "可以", "应该", "必须", "愿意", "同意", "拒绝",
        # Adjectives
        "强大", "聪明", "美丽", "危险", "重要", "简单", "复杂",
        "特殊", "普通", "正常", "奇怪", "可怕", "厉害",
        # Abstract nouns
        "问题", "情况", "方式", "方法", "原因", "结果", "过程",
        "时间", "地方", "事情", "消息", "信息", "数据", "能力",
        # Sci-fi generic
        "系统", "任务", "奖励", "等级", "能量", "攻击", "防御",
        "技能", "装备", "武器", "战斗", "战力",
    }

    # Patterns thường xuyên xuất hiện KHÔNG phải entity dù có suffix
    FALSE_POSITIVE_PATTERNS = [
        r"^[\d]+[级品阶]$",          # Thuần số + rank suffix
        r"^[这那此彼][\u4e00-\u9fff]",  # Demonstrative + noun
        r"^[很非常极]",              # Degree adverb đầu
        r"^[的地得]",               # Particles
        r"^[了着过]",               # Aspect markers
    ]

    def _apply_penalties(self, candidate: EntityCandidate,
                         text: str) -> float:
        score = candidate.confidence

        # Blacklist check
        if candidate.form in self.COMMON_WORD_BLACKLIST:
            return 0.0  # Hard reject

        # False positive pattern check
        for fp_pattern in self.FALSE_POSITIVE_PATTERNS:
            if re.match(fp_pattern, candidate.form):
                score -= 0.40

        # Single char penalty
        if len(candidate.form) == 1:
            score -= 0.80

        # Too generic (only suffix, no specific prefix)
        if candidate.form in {"军团", "舰队", "文明", "帝国", "组织"}:
            score -= 0.30

        # Preceded by 一个/一名/一位 → likely common noun not proper
        generic_det = rf"(?:一个|一名|一位|一只|一条)\s*{re.escape(candidate.form)}"
        if re.search(generic_det, text):
            score -= 0.20

        return max(score, 0.0)
```

### B.3 — PASS 5 bổ sung: Phoneme Rebuild Post-processing

```python
class TransliterationEngine:

    def _phoneme_rebuild(self, form: str) -> str:
        """
        Phiên âm ngược từ Hán sang Latin approximate.
        
        Rules:
        1. Ghép phoneme từng chữ
        2. Title-case toàn từ
        3. Xử lý các suffix đặc biệt
        4. Gộp nguyên âm cuối + đầu để tránh double vowel
        """
        phonemes = []
        for char in form:
            p = self.TRANSLIT_PHONEME_MAP.get(char, "")
            if p:
                phonemes.append(p)
            else:
                # Unknown char → dùng pinyin approximation
                phonemes.append(char)

        if not phonemes:
            return form

        # Post-process: gộp và capitalize
        result = "".join(phonemes)

        # Xử lý double vowel (aei → ai, oua → oa)
        result = re.sub(r"([aeiou])\1+", r"\1", result)

        # Capitalize đúng cách
        return result.title()

    def _is_transliteration(self, form: str) -> bool:
        """
        Phát hiện xem span có phải là tên phiên âm không.
        
        Heuristics:
        1. Không có nghĩa khi ghép lại (kiểm tra trong từ điển thông dụng)
        2. Kết hợp các phoneme phổ biến dùng cho phiên âm
        3. Không match common Chinese word patterns
        4. Độ dài 2-5 chữ
        """
        if len(form) < 2 or len(form) > 6:
            return False
        if form in self.COMMON_WORD_BLACKLIST:
            return False

        # Đếm số chars thuộc phoneme high-frequency set
        HIGH_FREQ_TRANSLIT = set(self.TRANSLIT_PHONEME_MAP.keys())
        translit_ratio = sum(1 for c in form if c in HIGH_FREQ_TRANSLIT) / len(form)
        return translit_ratio >= 0.6

    def get_vi_form(self, entity: EntityCandidate,
                    glossary: dict,
                    register: str = "modern") -> str:
        """
        Unified entry point:
        - Kiểm tra glossary trước
        - Quyết định METHOD A/B/C
        - Apply register rules
        """
        form = entity.form

        # METHOD C: Glossary (highest priority)
        if form in glossary:
            return glossary[form]

        # Sci-fi franchise lookup
        for franchise_dict in [
            self.MARVEL_GLOSSARY, self.DC_GLOSSARY,
            self.STARWARS_GLOSSARY, self.TRANSFORMERS_GLOSSARY
        ]:
            if form in franchise_dict:
                return franchise_dict[form]

        # METHOD B: Transliteration
        if form in self.TRANSLIT_CHAR_MAP:
            return self.TRANSLIT_CHAR_MAP[form]
        if self._is_transliteration(form):
            rebuilt = self._phoneme_rebuild(form)
            return rebuilt

        # METHOD A: Han-Viet
        return self._hanviet_convert(form, register)

    def _hanviet_convert(self, form: str, register: str) -> str:
        """
        Chuyển đổi Hán → Hán Việt.
        
        Register-aware:
        - xianxia: giữ nguyên Hán Việt (Lâm Động, Tu Vi)
        - sci-fi modern: có thể semantic-translate nếu tên chức danh
        - classical: Hán Việt thuần
        """
        # Dùng table lookup đã có trong dict
        # Fallback: transliterate từng chữ theo bảng Hán Việt
        result = []
        for char in form:
            hv = HAN_VIET_TABLE.get(char, char)
            result.append(hv)
        return " ".join(result).title()
```

```python
class AliasHierarchyResolver:

    def resolve_conflicts(self, edges: list[AliasEdge]) -> list[AliasEdge]:
        """
        Khi alias A→B và alias A→C xuất hiện cùng lúc:
        1. Nếu B và C là cùng loại (cả hai là EVOLUTION) → giữ cả hai
           như parallel evolution state
        2. Nếu confidence khác nhau → keep higher
        3. Nếu confidence bằng nhau → keep later chapter (more recent)
        4. Nếu một là EVOLUTION và một là NICKNAME → keep cả hai,
           với priority EVOLUTION > NICKNAME cho canonical lookup
        
        Special case: Multi-level protagonist alias
        李宇 → 龙尊 → 超越者 → 圣·天尊
        Tất cả point về canonical 李宇 (first seen name)
        """
        # Group by alias target
        by_alias: dict[str, list[AliasEdge]] = defaultdict(list)
        for edge in edges:
            by_alias[edge.alias].append(edge)

        resolved = []
        for alias, competing in by_alias.items():
            if len(competing) == 1:
                resolved.append(competing[0])
            else:
                # Multiple canonical forms for same alias
                # Sort by: confidence DESC, chapter DESC
                competing.sort(key=lambda e: (e.confidence, e.chapter_first),
                               reverse=True)
                # Keep best, mark others as secondary
                best = competing[0]
                best.is_primary = True
                resolved.append(best)
                for secondary in competing[1:]:
                    secondary.is_primary = False
                    secondary.conflict_with = best.canonical
                    resolved.append(secondary)

        return resolved

    def build_protagonist_chain(self,
                                 aliases: dict[str, str],
                                 protagonist_seed: dict) -> dict:
        """
        Xây dựng alias chain đặc biệt cho nhân vật chính.
        
        Input protagonist_seed:
        {
            "李宇": {"type": "PERSON", "role": "protagonist"},
            "aliases": ["龙尊", "超越者", "圣·天尊", "一号", "一号分身的本体"]
        }
        
        Output:
        Mọi alias đều resolve về 李宇 với VI form "Lý Vũ"
        """
        canonical = protagonist_seed["canonical"]
        vi_form = protagonist_seed["vi_form"]
        result = {canonical: vi_form}
        for alias in protagonist_seed.get("aliases", []):
            result[alias] = vi_form
        return result
```

```python
class ReviewExporter:

    def export_with_diff(self, records: list[EntityRecord],
                         existing_glossary_path: str,
                         output_path: str):
        """
        So sánh glossary mới với glossary cũ (nếu có).
        Output thêm file: changes.md với danh sách thêm/sửa/xóa.
        """
        old_glossary = {}
        if os.path.exists(existing_glossary_path):
            with open(existing_glossary_path) as f:
                old_glossary = json.load(f)

        new_glossary = {r.canonical: r.vi_form
                        for r in records if r.confidence >= 0.85}

        added   = {k: v for k, v in new_glossary.items()
                   if k not in old_glossary}
        removed = {k: v for k, v in old_glossary.items()
                   if k not in new_glossary}
        changed = {k: (old_glossary[k], new_glossary[k])
                   for k in new_glossary
                   if k in old_glossary and old_glossary[k] != new_glossary[k]}

        with open(f"{output_path}/changes.md", "w") as f:
            f.write(f"# Glossary Changes\n\n")
            f.write(f"## Added ({len(added)})\n")
            for k, v in sorted(added.items()):
                f.write(f"- `{k}` → `{v}`\n")
            f.write(f"\n## Removed ({len(removed)})\n")
            for k, v in sorted(removed.items()):
                f.write(f"- `{k}` (was `{v}`)\n")
            f.write(f"\n## Changed ({len(changed)})\n")
            for k, (old, new) in sorted(changed.items()):
                f.write(f"- `{k}`: `{old}` → `{new}`\n")

class EntityMemoryStore:
    """
    Lưu trữ entity memory persistent qua các chapter batch.
    Dùng SQLite để đảm bảo không mất data giữa các lần chạy.
    """

    def __init__(self, db_path: str):
        self.conn = sqlite3.connect(db_path)
        self._init_schema()

    def _init_schema(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS entities (
                canonical TEXT PRIMARY KEY,
                vi_form TEXT,
                entity_type TEXT,
                confidence REAL,
                first_chapter INTEGER,
                last_chapter INTEGER,
                chapter_count INTEGER,
                aliases TEXT,  -- JSON array
                sources TEXT   -- JSON array
            )
        """)
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS alias_map (
                alias TEXT PRIMARY KEY,
                canonical TEXT,
                alias_type TEXT,
                confidence REAL
            )
        """)
        self.conn.commit()

    def upsert_entity(self, record: EntityRecord):
        existing = self.conn.execute(
            "SELECT * FROM entities WHERE canonical = ?",
            (record.canonical,)
        ).fetchone()

        if existing:
            # Update: tăng chapter count, update last chapter
            self.conn.execute("""
                UPDATE entities SET
                    last_chapter = MAX(last_chapter, ?),
                    chapter_count = chapter_count + 1,
                    confidence = MAX(confidence, ?)
                WHERE canonical = ?
            """, (record.first_chapter, record.confidence, record.canonical))
        else:
            self.conn.execute("""
                INSERT INTO entities VALUES (?,?,?,?,?,?,?,?,?)
            """, (
                record.canonical, record.vi_form, record.type,
                record.confidence, record.first_chapter, record.first_chapter,
                1, json.dumps(record.aliases), json.dumps(record.sources)
            ))
        self.conn.commit()

    def get_alias(self, form: str) -> str | None:
        """Tra cứu alias → canonical → vi_form"""
        row = self.conn.execute(
            "SELECT canonical FROM alias_map WHERE alias = ?", (form,)
        ).fetchone()
        if not row:
            return None
        canonical = row[0]
        entity = self.conn.execute(
            "SELECT vi_form FROM entities WHERE canonical = ?", (canonical,)
        ).fetchone()
        return entity[0] if entity else None
```

### C.1 — GAP-J: `为...所...` và `被X所V` Classical Passive

```python
# File: src/grammar/transfer_rules/rule_wei_suo_passive.py

WEI_SUO_PATTERNS = [
    # 为X所V (classical: bị X làm V)
    # 为众人所知 → được mọi người biết đến
    # 为敌军所俘 → bị quân địch bắt giữ
    {
        "pattern": r"为([\u4e00-\u9fff]{1,8})所([\u4e00-\u9fff]{1,4})",
        "type": "wei_suo",
        "template": "{patient} bị/được {agent} {verb}",
    },
    # 被X所V (mixed classical-modern)
    # 被历史所铭记 → được lịch sử ghi nhớ
    # 被命运所捉弄 → bị số phận đùa giỡn
    {
        "pattern": r"被([\u4e00-\u9fff]{1,8})所([\u4e00-\u9fff]{1,4})",
        "type": "bei_suo",
        "template": "{patient} bị/được {agent} {verb}",
    },
]

def rule_wei_suo_passive(construction, nodes, vi_tokens):
    """
    为/被 + AGENT + 所 + V

    Decision bị vs được:
    - Dùng sentiment_lexicon của V (kế thừa từ BEI rule)
    - 为X所知/所见/所称 → được (positive)
    - 为X所杀/所俘/所困 → bị (negative)
    
    VI output:
    为众人所知 → được mọi người biết đến
    为命运所捉弄 → bị số phận đùa giỡn
    被历史所铭记 → được lịch sử ghi nhớ
    
    Xử lý 所:
    - 所 không được dịch (là marker cú pháp, không có nghĩa độc lập)
    - Mark 所 as skip_lexical
    """
    wei_idx = construction.marker_idx
    suo_idx = construction.suo_idx
    agent_span = construction.agent_span
    verb_idx = construction.verb_idx

    v_form = nodes[verb_idx].form
    v_sentiment = sentiment_lexicon.get(v_form, Sentiment.NEUTRAL)
    passive_marker = "được" if v_sentiment == Sentiment.POSITIVE else "bị"

    result = list(vi_tokens)
    # 为/被 → passive marker
    result[wei_idx] = VIToken(passive_marker)
    # 所 → skip
    result[suo_idx] = VIToken("", skip=True)

    return result, GrammarTrace(
        rule="wei_suo_passive",
        action=f"classical_passive_{passive_marker}",
        source_span=(wei_idx, verb_idx)
    )
```

### C.2 — GAP-L: Projection/Communication Scene

```python
# File: src/grammar/transfer_rules/rule_projection_scene.py

PROJECTION_INDICATORS = [
    "远程投影", "全息投影", "虚拟投影", "影像",
    "屏幕上", "投影仪", "远程会议", "通话",
    "连线", "视频通话", "传音", "传讯",
]

PHYSICAL_PRESENCE_INDICATORS = [
    "亲自", "本人", "现身", "到场", "出席",
    "进入会议室", "走进", "站在",
]

class ProjectionSceneDetector:
    """
    Phân biệt nhân vật hiện diện vật lý vs xuất hiện qua màn hình/hologram.
    
    Tại sao quan trọng:
    - Physical: "他走向XXX握手" → thực sự bắt tay
    - Projection: "他的投影走向XXX" → không thực sự
    - Ảnh hưởng đến dịch verb: "握手" → "bắt tay" vs "như thể bắt tay"
    
    Scope: paragraph-level detection
    """

    def detect_scene_type(self, paragraph: str) -> str:
        """Returns: "physical" | "projection" | "mixed" | "unknown" """
        proj_score = sum(1 for ind in PROJECTION_INDICATORS
                         if ind in paragraph)
        phys_score = sum(1 for ind in PHYSICAL_PRESENCE_INDICATORS
                         if ind in paragraph)
        if proj_score > 0 and phys_score == 0:
            return "projection"
        elif phys_score > 0 and proj_score == 0:
            return "physical"
        elif proj_score > 0 and phys_score > 0:
            return "mixed"
        return "unknown"

    def annotate_entities(self, paragraph: str,
                          entities: list[str]) -> dict[str, str]:
        """
        Trả về dict entity → presence_type
        "第一时间双方都没有说话" trong projection scene:
        → biết đây là projection → không dịch action verbs theo nghĩa literal
        """
        scene = self.detect_scene_type(paragraph)
        result = {}
        for entity in entities:
            if entity in paragraph:
                if f"{entity}的投影" in paragraph or f"{entity}的影像" in paragraph:
                    result[entity] = "projection"
                else:
                    result[entity] = scene if scene != "unknown" else "physical"
        return result

PROJECTION_VERB_MAP = {
    # Khi entity là projection → dùng verb nhẹ hơn hoặc thêm marker
    "physical_to_projection": {
        "走向": "hình chiếu tiến về phía",
        "握手": "như thể bắt tay",
        "拥抱": "như thể ôm",
        "站在": "hiện diện (qua hình chiếu) tại",
    }
}
```

```python
                    rule_name="quoted_number_guard",
                    start=m.start(1), end=m.end(1),
                    priority=5,
                    action="quoted_speech"
```

### D.20 — GAP-NEW-20: Nested `A的B的C` — 3 Levels

```python
class NestedPossessionRule:
    """
    Xử lý chuỗi sở hữu lồng nhau 3 cấp.
    
    ZH: A的B的C
    VI: C của B của A  (đảo ngược hoàn toàn)
    
    Nhưng có exception khi A/B/C là entity:
    - 星耀帝国的德罗耶达的命令
      → Mệnh lệnh của Droieda thuộc Tinh Diệu Đế Quốc
      (KHÔNG: "lệnh của Droieda của Đế Quốc")
    
    - 李宇的分身的实力
      → Thực lực của phân thân của Lý Vũ  (OK, giữ nguyên đảo)
    
    Algorithm:
    1. Parse 的-chain: A[的]B[的]C
    2. Identify types: ENTITY, ROLE, ATTRIBUTE, ITEM
    3. Apply inversion: C của B của A
    4. Post-merge: nếu B là role của A → "B + A" (không dùng của)
       (德罗耶达的帝国 → Đế Quốc Droieda, không "Đế Quốc của Droieda")
    """

    def parse_de_chain(self, tokens: list[Token]) -> list[tuple]:
        """
        Returns: list of (span, type) tuples separated by 的
        Example: [(A, ENTITY), (B, ROLE), (C, ATTRIBUTE)]
        """
        de_positions = [i for i, t in enumerate(tokens) if t.form == "的"]
        if len(de_positions) < 2:
            return []  # Not 3-level

        # Split into segments
        segments = []
        prev = 0
        for pos in de_positions:
            seg = tokens[prev:pos]
            segments.append(seg)
            prev = pos + 1
        segments.append(tokens[prev:])  # Last segment

        return [(seg, self._classify_segment(seg)) for seg in segments]

    def _classify_segment(self, tokens: list[Token]) -> str:
        """Classify segment as ENTITY/ROLE/ATTRIBUTE/ITEM"""
        form = "".join(t.form for t in tokens)
        if form in entity_memory:
            return "ENTITY"
        for suffix in ROLE_SUFFIXES:
            if form.endswith(suffix):
                return "ROLE"
        return "ATTRIBUTE"

    def transfer(self, segments: list[tuple]) -> str:
        """
        Transfer 3-level possession to VI.
        A (ENTITY), B (ROLE), C (ATTRIBUTE)
        → C của B [+ merge rule] A
        """
        if len(segments) == 3:
            a, type_a = segments[0]
            b, type_b = segments[1]
            c, type_c = segments[2]

            a_vi = translate(a)
            b_vi = translate(b)
            c_vi = translate(c)

            if type_b == "ROLE" and type_a == "ENTITY":
                # Merge: B của A → "B A" (tên + tổ chức)
                b_a_merged = f"{b_vi} {a_vi}"
                return f"{c_vi} của {b_a_merged}"
            else:
                return f"{c_vi} của {b_vi} của {a_vi}"

        return None  # Fallback to standard DE rule
```

```python
    Xử lý các trường hợp lỗi trong entity scanner.

                             exception: Exception,
                             fallback_result: list) -> list:
        """
        Nếu một pass fail → log error, trả về fallback (kết quả rỗng)
        Không cho phép một pass fail crash toàn pipeline.
        """
        logger.error(f"Entity scanner PASS {pass_name} failed: {exception}")
        return fallback_result  # Empty list, pipeline continues

    def handle_transliteration_unknown(self, form: str) -> str:
        """
        Khi không biết transliterate form như thế nào:
        1. Thử phoneme rebuild
        2. Nếu rebuild ra chuỗi không đọc được → giữ nguyên Hán Việt
        3. Flag for manual review
        """
        rebuilt = self._phoneme_rebuild(form)
        if not self._is_readable_latin(rebuilt):
            self._flag_for_review(form, "unresolvable_transliteration")
            return hanviet_convert(form)
        return rebuilt

    def handle_alias_cycle(self, chain: list[str]) -> str:
        """
        Khi alias chain tạo vòng tròn: A→B→C→A
        → Break cycle tại điểm đầu tiên
        → Log warning
        → Return first element as canonical
        """
        logger.warning(f"Alias cycle detected: {chain}")
        return chain[0]  # First seen = canonical

    def handle_confidence_tie(self, candidates: list[EntityCandidate]) -> EntityCandidate:
        """
        Khi 2 candidates có cùng confidence:
        → Chọn candidate xuất hiện trước (chapter nhỏ hơn)
        → Nếu cùng chapter → chọn cái dài hơn (longer form = more specific)
        """
        candidates.sort(key=lambda c: (c.chapter, -len(c.form)))
        return candidates[0]
```

### G.1 — Trie Cache Warm-up

```python
class TrieCacheManager:
    """
    Warm up Trie cache trước khi dịch để tránh cold-start penalty.
    """

    def warmup(self, entity_glossary: dict):
        """
        Load tất cả entity từ glossary vào Trie với priority cao.
        Chạy 1 lần khi khởi động.
        """
        for zh_form, vi_form in entity_glossary.items():
            self.trie.insert(zh_form, vi_form, priority=10)

    def batch_insert(self, records: list[EntityRecord]):
        for r in records:
            self.trie.insert(r.canonical, r.vi_form, priority=8)
            for alias in r.aliases:
                self.trie.insert(alias, r.vi_form, priority=7)
```

### G.2 — Rule Short-circuit

```python
class TransferEngine:
    def transfer(self, nodes, constructions, vi_tokens, registry):
        """
        Short-circuit: Nếu không có construction nào phù hợp với rule
        → skip rule hoàn toàn, không tốn overhead.
        """
        construction_types = {c.type for c in constructions}

        for rule in self.rules:
            # Short-circuit check
            if rule.trigger not in construction_types:
                continue  # Skip this rule entirely

            relevant = [c for c in constructions if c.type == rule.trigger]
            for construction in relevant:
                span = rule.get_primary_span(construction)
                if span and not registry.can_claim(*span, rule.priority):
                    continue

                vi_tokens, trace = rule.transform(
                    construction, nodes, vi_tokens, registry
                )

        return vi_tokens
```

```python
RULE_PERFORMANCE_BUDGET = {
    # Max milliseconds per rule per sentence
    "de_inversion":    5.0,   # Complex, allow more
    "nested_de":       8.0,   # Very complex
    "aspect":          2.0,   # Simple substitution
    "bei_passive":     2.0,
    "wei_suo":         2.0,
    "ba_construction": 4.0,
    "serial_verb":     5.0,
    "time_skip":       1.0,
    "dramatic_speech": 1.0,
    "meme_typography": 0.5,
    # Default budget for all others
    "default":         3.0,
}

class RulePerformanceMonitor:
    def run_with_budget(self, rule, *args) -> tuple:
        start = time.perf_counter()
        result = rule.transform(*args)
        elapsed_ms = (time.perf_counter() - start) * 1000

        budget = RULE_PERFORMANCE_BUDGET.get(rule.name,
                 RULE_PERFORMANCE_BUDGET["default"])

        if elapsed_ms > budget:
            logger.warning(
                f"Rule {rule.name} exceeded budget: "
                f"{elapsed_ms:.1f}ms > {budget}ms"
            )
        return result
```
