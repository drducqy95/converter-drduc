### 1.2 PASS 0 — Chapter Title Pre-scan

**Lý do:** Tiêu đề chương là nguồn dữ liệu entity chất lượng cao nhất — tác giả đã chủ ý đặt entity quan trọng vào đó.

```python
class ChapterTitleEntitySeeder:
    """
    Quét toàn bộ tiêu đề chương để seed entity memory
    trước khi bắt đầu dịch bất kỳ chương nào.
    
    Input:  list[tuple[int, str]]  # [(chapter_num, title_text)]
    Output: dict[str, EntitySeed]
    """

    def scan_all_titles(self, chapters: list[tuple[int, str]]) -> dict:
        seeds = {}
        for ch_num, title in chapters:
            # Chiết xuất bracket items
            bracket_items = self._extract_brackets(title)
            for item in bracket_items:
                seeds[item] = EntitySeed(
                    form=item, source="chapter_title",
                    chapter=ch_num, confidence=0.95,
                    kind="BRACKETED_ITEM"
                )
            # Chiết xuất proper nouns (non-bracket)
            proper_nouns = self._extract_proper_nouns(title)
            for noun in proper_nouns:
                seeds[noun] = EntitySeed(
                    form=noun, source="chapter_title",
                    chapter=ch_num, confidence=0.80,
                    kind="PROPER_NOUN_CANDIDATE"
                )
        return seeds

    def _extract_brackets(self, text: str) -> list[str]:
        """
        Chiết xuất nội dung trong 【...】 và [...]
        Xử lý nested: 【获得白色称号【菊花守护者】】
        → inner: 菊花守护者
        → outer: 获得白色称号【菊花守护者】
        """
        results = []
        # Nested bracket parser
        stack = []
        i = 0
        while i < len(text):
            if text[i] == "【":
                stack.append(i)
            elif text[i] == "】" and stack:
                start = stack.pop()
                inner = text[start+1:i]
                results.append(inner)
            i += 1
        return results

    def _extract_proper_nouns(self, text: str) -> list[str]:
        """
        Sau khi xóa bracket content, tìm proper noun candidates:
        1. Cụm viết hoa (nếu có)
        2. Cụm với known suffix (舰队/文明/帝国/族/星...)
        3. Transliteration patterns (tên phiên âm Latin→Hán)
        """
        # Remove brackets first
        clean = re.sub(r"【[^】]*】", "", text)
        candidates = []
        # Suffix-based detection
        for suffix in ENTITY_SUFFIX_LIST:
            pattern = rf"[\u4e00-\u9fff]{{1,8}}{re.escape(suffix)}"
            matches = re.findall(pattern, clean)
            candidates.extend(matches)
        # Transliteration detection
        translit = self._detect_transliteration(clean)
        candidates.extend(translit)
        return list(set(candidates))
```

### 1.3 PASS 1 — Bracket Entity Extraction

```python
class BracketItemExtractor:
    """
    Chiết xuất và phân loại entity từ 【...】 trong body text.
    Khác PASS 0 ở chỗ: xử lý body paragraph, không chỉ tiêu đề.
    """

    # Loại bracket item
    KIND_CLASSIFIERS = [
        # System panel
        (r"^宿主[:：]", "SYSTEM_HOST"),
        (r"^分身", "SYSTEM_CLONE"),
        (r"^获得", "SYSTEM_REWARD"),
        (r"^[冷却|回收|投放|结算|升级]", "SYSTEM_STATE"),
        (r"^第\d+天[:：]", "SYSTEM_DAY_LOG"),
        # Item/ability
        (r"[天赋|天才|才能]$", "ABILITY_TALENT"),
        (r"[称号|头衔]", "TITLE_HONORIFIC"),
        (r"[基因链|殖装|血清|药剂]$", "BIO_ITEM"),
        (r"[手套|宝石|权杖|王冠|刀|枪|剑|盾]$", "ARTIFACT"),
        (r"[死星|帝死星]$", "SUPER_WEAPON"),
        (r"[感应|魔法|附魔|无效化|打造]$", "ABILITY"),
        # Entity
        (r"世界$", "WORLD"),
        (r"[文明|帝国|联盟]$", "POLITY"),
    ]

    def extract(self, text: str) -> list[BracketEntity]:
        results = []
        for match in re.finditer(r"【([^】]+)】", text):
            content = match.group(1)
            kind = self._classify(content)
            results.append(BracketEntity(
                form=content,
                kind=kind,
                start=match.start(),
                end=match.end(),
                protected=True
            ))
        return results

    def _classify(self, content: str) -> str:
        for pattern, kind in self.KIND_CLASSIFIERS:
            if re.search(pattern, content):
                return kind
        return "UNKNOWN_ITEM"
```

### 1.4 PASS 2 — Pattern-based NER

```python
class PatternBasedNER:
    """
    Nhận diện entity dựa trên pattern ngôn ngữ học.
    Không dùng ML model — pure rule + pattern.
    
    Các loại pattern:
    TYPE 1: Suffix-based (80% accuracy)
    TYPE 2: Prefix-based
    TYPE 3: Transliteration-based (tên Latin phiên âm ra Hán)
    TYPE 4: Role-title-based
    TYPE 5: Structural pattern (A的B, A是B, A称为B)
    """

    # ── TYPE 1: SUFFIX-BASED ──────────────────────────────────────
    SUFFIX_ONTOLOGY = {
        # Person
        "PERSON": ["者", "人", "主", "王", "帝", "尊", "君", "师",

                   "舰长", "指挥官", "议长", "族长", "团长"],

        "SPECIES": ["族", "种族", "生命体", "异形", "克隆体",

        # Item/Tech
        "ARTIFACT": ["宝石", "手套", "权杖", "王冠", "法宝",
                     "宝典", "神器"],
        "WEAPON": ["刀", "剑", "枪", "手枪", "炮", "弓", "矛"],
        "EQUIPMENT": ["殖装", "装甲", "护盾", "外骨骼"],

        "SHIP": ["飞船", "战舰", "母舰", "旗舰", "运输舰",

    # Tên phiên âm từ Latin: thường là 2-5 chữ ghép từ limited phonemes
    TRANSLITERATION_PATTERNS = [
        # Đặc điểm: không có nghĩa Hán phổ biến khi ghép lại
        # Pattern: 2-5 chars, mỗi char thuộc phoneme set phổ biến dùng phiên âm
        r"[阿艾埃爱奥安昂]{1}[尔尔尔]{0,1}[\u4e00-\u9fff]{1,3}",
        r"[维威韦]{1}[\u4e00-\u9fff]{1,3}",
        r"[科克]{1}[\u4e00-\u9fff]{1,3}",
        r"[德迪]{1}[\u4e00-\u9fff]{1,3}",
        r"[莫默]{1}[\u4e00-\u9fff]{1,3}",
        r"[泰塔]{1}[\u4e00-\u9fff]{1,3}",
        r"[罗洛]{1}[\u4e00-\u9fff]{1,3}",
    ]

    # ── TYPE 4: ROLE-TITLE PATTERNS ───────────────────────────────
    ROLE_TITLE_MARKERS = {
        # Chức danh trước tên
        "pre_name": ["舰长", "将军", "议长", "指挥官", "大人",
                     "阁下", "先生", "女士", "殿下", "陛下",
                     "道友", "前辈", "后辈"],
        # Cụm "được gọi là X"
        "naming_verbs": ["名为", "称为", "叫做", "自称", "又叫",
                         "外号叫", "绰号"],
    }

    # ── TYPE 5: STRUCTURAL PATTERNS ───────────────────────────────
    NAMING_STRUCTURES = [
        # 名为X的Y — Y tên là X
        (r"名为([^，。！？\s]{2,8})的[\u4e00-\u9fff]{1,4}", 1),
        # X叫做Y — X tên là Y
        (r"[\u4e00-\u9fff]{1,4}叫做([^，。！？\s]{2,6})", 1),
        # 那个X / 这个X khi X là tên
        (r"[那这]个([A-Z\u4e00-\u9fff]{2,6})", 1),
        # X阁下 / X将军 / X陛下 — kính ngữ sau tên
        (r"([A-Z\u4e00-\u9fff]{2,6})(阁下|将军|陛下|殿下|大人)", 1),
    ]

    def scan(self, text: str, chapter: int,
             known_entities: dict) -> list[EntityCandidate]:
        candidates = []

        # TYPE 1: Suffix scan
        for entity_type, suffixes in self.SUFFIX_ONTOLOGY.items():
            for suffix in suffixes:
                pattern = rf"[\u4e00-\u9fff]{{1,8}}{re.escape(suffix)}"
                for m in re.finditer(pattern, text):
                    span = m.group()
                    if span not in known_entities:
                        candidates.append(EntityCandidate(
                            form=span, type=entity_type,
                            source="suffix_pattern",
                            chapter=chapter, confidence=0.70
                        ))

        # TYPE 3: Transliteration detection
        for pattern in self.TRANSLITERATION_PATTERNS:
            for m in re.finditer(pattern, text):
                candidates.append(EntityCandidate(
                    form=m.group(), type="PERSON_TRANSLIT",
                    source="transliteration_pattern",
                    chapter=chapter, confidence=0.55
                ))

        # TYPE 5: Structural patterns
        for pattern, group_idx in self.NAMING_STRUCTURES:
            for m in re.finditer(pattern, text):
                name = m.group(group_idx)
                candidates.append(EntityCandidate(
                    form=name, type="NAMED_ENTITY",
                    source="naming_structure",
                    chapter=chapter, confidence=0.75
                ))

        return candidates
```

### 1.5 PASS 3 — Context-based Confirmation

```python
class ContextConfirmationPass:
    """
    Dùng context quanh candidate để tăng/giảm confidence.
    
    Tăng confidence khi:
    - Candidate là subject của verb nhiều lần
    - Candidate xuất hiện sau speech verb (X说/X道)
    - Candidate xuất hiện sau kính ngữ (X阁下/X大人)
    - Candidate xuất hiện trong bracket 【X】
    - Candidate xuất hiện trong chapter title
    
    Giảm confidence khi:
    - Candidate là common adjective/verb khi đứng một mình
    - Candidate match common word list
    - Candidate length = 1 (single char, probably not entity)
    """

    CONFIDENCE_BOOSTS = {
        "is_bracketed":          +0.30,
        "in_chapter_title":      +0.25,
        "after_speech_verb":     +0.20,
        "after_honorific":       +0.25,
        "as_subject_repeated":   +0.15,  # per occurrence, cap at +0.45
        "in_dialogue_addressee": +0.20,
        "named_via_naming_verb": +0.25,
        "has_alias_link":        +0.20,
    }

    CONFIDENCE_PENALTIES = {
        "is_common_word":        -0.50,
        "length_one":            -0.80,
        "is_stop_word":          -1.00,
        "no_context_support":    -0.10,
    }

    SPEECH_VERBS = {
        "说", "道", "喊", "叫", "怒道", "笑道", "问道",
        "答道", "沉声道", "冷道", "低语", "暗道", "心道"
    }

    def confirm(self, candidate: EntityCandidate,
                full_text: str) -> EntityCandidate:
        score = candidate.confidence

        # Check bracketed
        if f"【{candidate.form}】" in full_text:
            score += self.CONFIDENCE_BOOSTS["is_bracketed"]

        # Count as subject
        subject_count = self._count_as_subject(candidate.form, full_text)
        score += min(subject_count * 0.15, 0.45)

        # After speech verb: X说/X道
        for sv in self.SPEECH_VERBS:
            if re.search(rf"{re.escape(candidate.form)}{sv}", full_text):
                score += self.CONFIDENCE_BOOSTS["after_speech_verb"]
                break

        # After honorific
        for h in ["阁下", "大人", "陛下", "殿下", "将军"]:
            if re.search(rf"{re.escape(candidate.form)}{h}", full_text):
                score += self.CONFIDENCE_BOOSTS["after_honorific"]
                break

        # Common word penalty
        if candidate.form in COMMON_CHINESE_WORDS:
            score += self.CONFIDENCE_PENALTIES["is_common_word"]

        candidate.confidence = min(max(score, 0.0), 1.0)
        return candidate

    def _count_as_subject(self, form: str, text: str) -> int:
        # Pattern: X + [是/有/将/会/要/可以/不得不] hoặc X + VV
        pattern = rf"{re.escape(form)}(?:是|有|将|会|要|可以|不得不|[\u4e00-\u9fff]{{1,2}}了)"
        return len(re.findall(pattern, text))
```

### 1.6 PASS 4 — Cross-chapter Cooccurrence

```python
class CrossChapterCooccurrence:
    """
    Entity xuất hiện ở nhiều chương → confidence cao hơn.
    Entity xuất hiện cùng các entity đã biết → confidence cao hơn.
    
    Algorithm:
    1. Map each candidate → set of chapter numbers
    2. Candidates in 3+ chapters get +0.15
    3. Candidates in 10+ chapters get +0.25
    4. Candidates cooccurring with known entities get +0.10
    """

    def analyze(self, candidates: list[EntityCandidate],
                known: dict) -> list[EntityCandidate]:
        # Count chapter appearances
        chapter_map: dict[str, set] = defaultdict(set)
        for c in candidates:
            chapter_map[c.form].add(c.chapter)

        for c in candidates:
            chapters = chapter_map[c.form]
            if len(chapters) >= 10:
                c.confidence += 0.25
            elif len(chapters) >= 3:
                c.confidence += 0.15
            elif len(chapters) >= 1:
                c.confidence += 0.05

            # Cooccurrence with known entities
            if c.form in known:
                c.confidence += 0.10

            c.confidence = min(c.confidence, 1.0)

        return candidates
```

```python
                      glossary: dict) -> str:

        # METHOD C: Franchise glossary first
        if form in glossary:
            return glossary[form]

        # METHOD B: Reverse transliteration (phoneme rebuild)
        if entity.source == "transliteration_pattern":
            if form in self.TRANSLIT_CHAR_MAP:
                return self.TRANSLIT_CHAR_MAP[form]
            # Fallback: rebuild from phoneme map
            return self._phoneme_rebuild(form)

        # METHOD A: Han-Viet
        return hanviet_convert(form)

    def _phoneme_rebuild(self, form: str) -> str:
        """
        Rebuild Latin approximation từ Hán phiên âm.
        Ví dụ: 科德里 → ke+de+li → Kodeli → Kodeli
        """
        result = []
        for char in form:
            phoneme = self.TRANSLIT_PHONEME_MAP.get(char, char)
            result.append(phoneme.capitalize() if not result else phoneme)
        return "".join(result).title()
```

### 1.8 PASS 6 — Alias Resolution

```python
class AliasHierarchyResolver:
    """
    Xây dựng và giải quyết alias chain.
    
    Alias types:
    TYPE 1: Title alias — nhân vật được gọi bằng chức danh thay tên
            李宇 → 龙尊 (khi trở thành Dragon)
    TYPE 2: Evolution alias — entity tiến hóa sang form mới
            死星 → 帝死星 (khi được upgrade)
    TYPE 3: Nickname — tên thân mật hoặc biệt danh
            三胖子 → 三先生 (hoặc tên thật)
    TYPE 4: Abbreviation — tên rút gọn
            大宇宙联盟 → 联盟
    TYPE 5: Cross-franchise — tên franchise sang tên VI
            钢铁侠 → Iron Man / Tony Stark
    
    Algorithm:
    1. Scan text for known entity forms
    2. When entity appears followed/preceded by 又叫/也叫/即/自称/被称为:
       extract alias pair
    3. When evolution event verb (进阶/升级/点化) connects two entities:
       create EVOLUTION alias
    4. Multi-level chain flattening:
       A→B→C means A aliases B and C
    """

    @dataclass
    class AliasEdge:
        canonical: str          # canonical form (first seen or highest confidence)
        alias: str              # alternative form
        alias_type: str         # TYPE 1-5
        chapter_first: int      # where alias first appeared
        confidence: float       # how confident this link is

    ALIAS_TRIGGER_VERBS = {
        "又叫": "NICKNAME",
        "也叫": "NICKNAME",
        "自称": "SELF_TITLE",
        "被称为": "EXTERNAL_TITLE",
        "即": "IDENTITY",
        "也就是": "IDENTITY",
        "说的是": "REFERENCE",
        "就是": "IDENTITY",
    }

    EVOLUTION_VERBS = {
        "点化": "EVOLUTION",
        "进阶": "EVOLUTION",
        "升华": "EVOLUTION",
        "升级": "UPGRADE",
        "诞生": "CREATION",
        "出世": "AWAKENING",
    }

    def build_alias_graph(self, text: str,
                          entities: dict,
                          chapter: int) -> list[AliasEdge]:
        edges = []

        # TYPE 1/2/3: Trigger verb detection
        for trigger, alias_type in self.ALIAS_TRIGGER_VERBS.items():
            pattern = rf"([\u4e00-\u9fff]{{2,8}}){trigger}([\u4e00-\u9fff]{{2,8}})"
            for m in re.finditer(pattern, text):
                a, b = m.group(1), m.group(2)
                if a in entities or b in entities:
                    canonical = a if a in entities else b
                    alias = b if a in entities else a
                    edges.append(self.AliasEdge(
                        canonical=canonical, alias=alias,
                        alias_type=alias_type, chapter_first=chapter,
                        confidence=0.85
                    ))

        # TYPE 2: Evolution alias
        for verb, ev_type in self.EVOLUTION_VERBS.items():
            pattern = rf"([\u4e00-\u9fff]{{2,8}}){verb}(.{{0,2}})([\u4e00-\u9fff]{{2,8}})"
            for m in re.finditer(pattern, text):
                source, target = m.group(1), m.group(3)
                if source in entities:
                    edges.append(self.AliasEdge(
                        canonical=source, alias=target,
                        alias_type=ev_type, chapter_first=chapter,
                        confidence=0.80
                    ))

        return edges

    def resolve_chain(self, edges: list[AliasEdge]) -> dict[str, str]:
        """
        Flatten multi-level chain:
        A→B, B→C → map C to canonical A
        
        Conflict resolution:
        If C also aliases D (different chain), keep higher-confidence link.
        """
        # Build adjacency
        graph: dict[str, list[AliasEdge]] = defaultdict(list)
        for edge in edges:
            graph[edge.canonical].append(edge)

        # BFS/DFS flatten
        canonical_map: dict[str, str] = {}
        for canonical in graph:
            visited = {canonical}
            queue = [canonical]
            while queue:
                node = queue.pop(0)
                for edge in graph.get(node, []):
                    if edge.alias not in visited:
                        canonical_map[edge.alias] = canonical
                        visited.add(edge.alias)
                        queue.append(edge.alias)

        return canonical_map
```

### 2.15 GAP-W: `并非/并没有/并不是` Emphatic Negation

```python
EMPHATIC_NEG_MAP = {
    "并非": "hoàn toàn không phải / chẳng phải",
    "并没有": "hoàn toàn không / chẳng có",
    "并不是": "hoàn toàn không phải",
    "并不": "hoàn toàn không / chẳng",
    "绝非": "tuyệt đối không phải",
    "绝不": "tuyệt đối không",
    "断然不是": "tuyệt đối không phải",
}

def rule_emphatic_negation(sentence: str) -> str:
    """
    他并非真正意义上的死亡
    → Hắn hoàn toàn không phải là đã chết theo nghĩa thực sự
    
    他并不是没有考虑过
    → Hắn hoàn toàn không phải là chưa từng suy nghĩ
    
    Note: "不会真正意义上的死亡" = V-NEG-COMPLEMENT-FULL-MEANING
    → "sẽ không chết theo đúng nghĩa của nó"
    """
```
